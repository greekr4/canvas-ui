젤리 슬라이더

진짜 <input type=range>를 촬영해, 드래그 속도에 따라 스프링 물리로 출렁이는 젤리 카드. 값에 따라 색조가 변합니다.

커스텀

사용법

const jelly = CanvasUI.create({
  mount: '#preview',
  html: `
    <div class="js-card">
      <div class="js-value-row">
        <span class="js-value">50</span><span class="js-unit">%</span>
      </div>
      <input class="js-range" type="range" min="0" max="100" value="50">
      <div class="js-hint">썸을 드래그해 보세요</div>
    </div>`,
  width: 480, height: 280,
  onFrame(t, api) { /* 스프링 물리: 값 추적 + 속도 기반 변형량 계산 */ },
  effect(ctx, src, t, api) { /* 촬영본을 squash&stretch로 변형 복사 */ },
  onHit(el, ev, api) { /* range를 누르면 커스텀 드래그 시작, true 반환 */ },
});
