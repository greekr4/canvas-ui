3D 키보드

한/영 키캡 DOM 키보드를 원근 기울기로 렌더합니다. 물리 키보드를 치거나 키캡을 클릭하면 키가 눌리고, 그 위치에서 잉크 리플이 canvas로 퍼집니다. 입력은 키보드 위의 진짜 input에 쌓입니다.

커스텀

사용법

const kb = CanvasUI.create({ mount: '#mount', html: keyboardHtml, // .kbd3d 마크업 — input + button[data-key] 키캡 width: 760, height: 440, onFrame: (t, api) => { // 리플 진행 (물리 갱신은 effect 앞, onFrame에서) api.state.ripples = (api.state.ripples || []).filter(r => r.alpha > 0.02); api.state.ripples.forEach(r => { r.r += 5; r.alpha *= 0.93; }); }, effect: (ctx, src, t, api) => { // 소스 canvas를 가로 스트립으로 잘라 폭을 보간 → 원근 사다리꼴 (유사 3D) const g = geom(api); // tiltDeg → scaleTop·squash·중심 좌표 for (let sy = 0; sy < g.H; sy += 3) { const s = g.scaleTop + (1 - g.scaleTop) * (sy / g.H); ctx.drawImage(src, 0, sy, g.W, 3, g.cx - g.W * s / 2, g.topY + sy * g.squash, g.W * s, 3 * g.squash + 1); } // 리플은 같은 원근 매핑으로 타원 링을 그린다 }, onHit: (el, ev, api) => { // effect가 좌표를 변형하므로 역변환 후 api.hit()으로 직접 판정 const local = unproject(api, ev); // fx 좌표 → 기울기 역변환 → 로컬 const t2 = api.hit(local.x, local.y); const btn = t2 && t2.closest('button[data-key]'); if (btn) { flash(btn); btn.click(); } // click → data-key 문자를 input에 삽입 return true; // 기본 포워딩(미보정 좌표) 생략 }, }); kb.state.tiltDeg = 22; // 커스텀: 기울기 각도 (0~40) kb.state.rippleColor = '#6366f1'; // 커스텀: 리플 색
